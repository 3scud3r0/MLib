# Contribuir com LeanML

Use o compilador fixado em `lean-toolchain` e as revisões de `lake-manifest.json`.
Não substitua hipóteses ausentes por axiomas, `sorry`, `admit` ou avaliação nativa.
Use `module`, `public import` e `@[expose] public section`, conforme o sistema
de módulos do Lean fixado no projeto. Prefira definições existentes do Mathlib e imports específicos. Teoremas reutilizados
devem manter as hipóteses e a atribuição ao Mathlib claramente documentadas.

Para cada mudança matemática:

1. Declare domínio, normalização, integrabilidade, suporte e mensurabilidade necessários.
2. Acrescente uma aplicação concreta ou teste de regressão em `LeanMLTest/Basic.lean` ou `LeanMLTest/Advanced.lean`.
3. Liste os resultados principais em `LeanMLTest/Axioms.lean`.
4. Execute `bash scripts/check.sh` e atualize `docs/STATUS.md`.

O script verifica ausência de lacunas, compila a biblioteca e os exemplos e
inspeciona os axiomas dos principais teoremas. Os axiomas lógicos usuais de Lean
(`propext`, `Classical.choice`, `Quot.sound`) são permitidos; lacunas e dependência
da avaliação nativa não são aceitas como provas.

Resultados ainda não demonstrados podem ser definidos como propriedades em `Prop`,
com status explícito de especificação. Não crie um teorema que apenas recebe como
hipótese a conclusão desejada e o anuncie como demonstração do resultado original.

O escopo ativo é aprendizado de máquina formalizado. Não reintroduza os módulos
Python/Rust históricos ou seus workflows na árvore ativa.

public import Mathlib.Analysis.Normed.Module.Basic
@[expose] public section
namespace LeanML.GameTheory
variable {E F : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E]
  [NormedAddCommGroup F] [NormedSpace ℝ F]

/-- Sion's minimax theorem on two nonempty compact convex strategy sets.
This is the saddle-point formulation, reused from Mathlib's proof of Sion.
Neither a nonconvex GAN nor its training algorithm satisfies these hypotheses automatically.
-/
theorem minimax_saddle {X : Set E} {Y : Set F} {payoff : E → F → ℝ}
    (hneX : X.Nonempty) (hconvX : Convex ℝ X) (hcompactX : IsCompact X)
    (hneY : Y.Nonempty) (hconvY : Convex ℝ Y) (hcompactY : IsCompact Y)
    (hlower : ∀ y ∈ Y, LowerSemicontinuousOn (fun x => payoff x y) X)
    (hconvex : ∀ y ∈ Y, QuasiconvexOn ℝ X (fun x => payoff x y))
    (hupper : ∀ x ∈ X, UpperSemicontinuousOn (fun y => payoff x y) Y)
    (hconcave : ∀ x ∈ X, QuasiconcaveOn ℝ Y (fun y => payoff x y)) :
    ∃ x ∈ X, ∃ y ∈ Y, IsSaddlePointOn X Y payoff x y :=
  Sion.exists_isSaddlePointOn hneX hconvX hcompactX hlower hconvex
    hconvY hneY hcompactY hupper hconcave
end LeanML.GameTheory

namespace LeanML.Generative

/-- Variational lower bound in its KL-gap form. A probabilistic instantiation
must additionally derive the ELBO/KL decomposition from its densities.
-/
def elboFromGap (logEvidence posteriorKL : ℝ) : ℝ := logEvidence-posteriorKL

/-- Any nonnegative variational gap yields a lower bound on log evidence. -/
theorem elbo_le_logEvidence (logEvidence posteriorKL : ℝ) (hKL : 0 ≤ posteriorKL) :
    elboFromGap logEvidence posteriorKL ≤ logEvidence := sub_le_self _ hKL

/-- The bound is tight exactly when the variational gap vanishes. -/
theorem elbo_eq_iff (logEvidence posteriorKL : ℝ) :
    elboFromGap logEvidence posteriorKL = logEvidence ↔ posteriorKL = 0 := by
  unfold elboFromGap
  constructor <;> intro h <;> linarith
end LeanML.Generative

-- The following finite-latent ELBO is derived directly from Jensen's inequality.

noncomputable section
open scoped BigOperators
namespace LeanML.Generative
open LeanML.Information
variable {Z : Type*} [Fintype Z]

/-- Actual finite-latent ELBO from positive joint likelihoods and a variational law. -/
def finiteELBO (joint : Z → ℝ) (q : FiniteDistribution Z) : ℝ :=
  ∑ z, q.mass z*Real.log (joint z/q.mass z)

/-- Jensen's inequality proves the variational bound with all positivity assumptions explicit. -/
theorem finiteELBO_le_logEvidence (joint : Z → ℝ) (q : FiniteDistribution Z)
    (hjoint : ∀ z, 0 < joint z) (hq : ∀ z, 0 < q.mass z) :
    finiteELBO joint q ≤ Real.log (∑ z, joint z) := by
  have h := strictConcaveOn_log_Ioi.concaveOn.le_map_sum
    (t := Finset.univ) (w := q.mass) (p := fun z => joint z/q.mass z)
    (fun z _ => q.nonneg z) q.total (fun z _ => div_pos (hjoint z) (hq z))
  have hs : ∀ z, q.mass z*(joint z/q.mass z) = joint z := by
    intro z
    field_simp [ne_of_gt (hq z)]
  simpa [finiteELBO, smul_eq_mul, hs] using h
end LeanML.Generative

variable {X Y : Type*} [Fintype X] [Fintype Y]

/-- A finite Kantorovich coupling with both marginal constraints enforced. -/
structure Coupling (P : FiniteDistribution X) (Q : FiniteDistribution Y) where
  mass : X → Y → ℝ
  nonneg : ∀ x y, 0 ≤ mass x y
  left_marginal : ∀ x, ∑ y, mass x y = P.mass x
  right_marginal : ∀ y, ∑ x, mass x y = Q.mass y

def Coupling.cost {P : FiniteDistribution X} {Q : FiniteDistribution Y}
    (π : Coupling P Q) (c : X → Y → ℝ) : ℝ := ∑ x, ∑ y, π.mass x y*c x y

def independentCoupling (P : FiniteDistribution X) (Q : FiniteDistribution Y) : Coupling P Q where
  mass x y := P.mass x*Q.mass y
  nonneg x y := mul_nonneg (P.nonneg x) (Q.nonneg y)
  left_marginal x := by simp [← Finset.mul_sum, Q.total]
  right_marginal y := by simp [← Finset.sum_mul, P.total]

/-- The optimal cost is an infimum over actual couplings, not over arbitrary matrices. -/
def kantorovichCost (P : FiniteDistribution X) (Q : FiniteDistribution Y)
    (c : X → Y → ℝ) : ℝ := ⨅ π : Coupling P Q, π.cost c

theorem Coupling.cost_nonneg {P : FiniteDistribution X} {Q : FiniteDistribution Y}
    (π : Coupling P Q) (c : X → Y → ℝ) (hc : ∀ x y, 0 ≤ c x y) : 0 ≤ π.cost c := by
  exact Finset.sum_nonneg fun x _ => Finset.sum_nonneg fun y _ =>
    mul_nonneg (π.nonneg x y) (hc x y)
end LeanML.Generative

noncomputable section
open LeanML.Information
namespace LeanML.Generative
variable {X Y : Type*} [Fintype X] [Fintype Y]

/-- Nonnegative costs have nonnegative optimal transport cost. -/
theorem kantorovichCost_nonneg (P : FiniteDistribution X) (Q : FiniteDistribution Y)
    (c : X → Y → ℝ) (hc : ∀ x y, 0 ≤ c x y) : 0 ≤ kantorovichCost P Q c := by
  let : Nonempty (Coupling P Q) := ⟨independentCoupling P Q⟩
  exact le_ciInf fun π => π.cost_nonneg c hc
end LeanML.Generative

/-! # Diffusion interfaces
Time-reversal of diffusions requires analytic and stochastic hypotheses beyond
this interface. No reversal or convergence theorem is asserted here.
-/
noncomputable section
namespace LeanML.Generative

structure DiffusionCoefficients (E : Type*) where
  drift : ℝ → E → E
  diffusion : ℝ → E → E

/-- Weak convergence expressed through bounded continuous test functions. -/
def WeakConvergence {E : Type*} [MeasurableSpace E] [TopologicalSpace E] [OpensMeasurableSpace E]
    (laws : ℕ → MeasureTheory.ProbabilityMeasure E)
    (limit : MeasureTheory.ProbabilityMeasure E) : Prop :=
  Filter.Tendsto laws Filter.atTop (nhds limit)

/-- Reversal of a deterministic time-indexed path, before any stochastic law is specified. -/
def reversePath {E : Type*} (horizon : ℝ) (path : ℝ → E) (t : ℝ) : E := path (horizon-t)

theorem reversePath_involutive {E : Type*} (T : ℝ) (path : ℝ → E) :
    reversePath T (reversePath T path) = path := by
  funext t
  simp [reversePath]
end LeanML.Generative

module
public import LeanML.Generative.OptimalTransport
public import Mathlib.Topology.MetricSpace.Lipschitz
public import Mathlib.Tactic.Linarith
@[expose] public section
noncomputable section
open scoped BigOperators
namespace LeanML.Generative
open LeanML.Information
variable {X : Type*} [Fintype X] [PseudoMetricSpace X]

omit [PseudoMetricSpace X] in
/-- A coupling converts the marginal difference of expectations to a paired expectation. -/
theorem coupling_expectation_difference (P Q : FiniteDistribution X) (π : Coupling P Q)
    (f : X → ℝ) :
    (∑ x, P.mass x*f x)-(∑ y, Q.mass y*f y) = ∑ x, ∑ y, π.mass x y*(f x-f y) := by
  simp_rw [mul_sub, Finset.sum_sub_distrib]
  congr 1
  · apply Finset.sum_congr rfl
    intro x _
    rw [← Finset.sum_mul, π.left_marginal]
  · rw [Finset.sum_comm]
    apply Finset.sum_congr rfl
    intro y _
    rw [← Finset.sum_mul, π.right_marginal]

/-- Weak Kantorovich-Rubinstein duality for every finite coupling. -/
theorem lipschitz_potential_le_cost (P Q : FiniteDistribution X) (π : Coupling P Q)
    (f : X → ℝ) (hf : LipschitzWith 1 f) :
    (∑ x, P.mass x*f x)-(∑ y, Q.mass y*f y) ≤ π.cost (fun x y => dist x y) := by
  rw [coupling_expectation_difference P Q π f]
  apply Finset.sum_le_sum
  intro x _
  apply Finset.sum_le_sum
  intro y _
  apply mul_le_mul_of_nonneg_left _ (π.nonneg x y)
  have h := hf.dist_le_mul x y
  have h' : |f x-f y| ≤ dist x y := by simpa only [NNReal.coe_one, one_mul, Real.dist_eq] using h
  exact (le_abs_self _).trans h'

/-- Every admissible dual potential lower-bounds the finite optimal cost. -/
theorem lipschitz_potential_le_optimal (P Q : FiniteDistribution X) (f : X → ℝ)
    (hf : LipschitzWith 1 f) :
    (∑ x, P.mass x*f x)-(∑ y, Q.mass y*f y) ≤ kantorovichCost P Q (fun x y => dist x y) := by
  let : Nonempty (Coupling P Q) := ⟨independentCoupling P Q⟩
  exact le_ciInf fun π => lipschitz_potential_le_cost P Q π f hf

/-- Unnormalized one-dimensional Gaussian mixture components, variance `v`. -/
def gaussianComponent (weight center : ℝ) (v x : ℝ) : ℝ :=
  weight*Real.exp (-((x-center)^2)/(2*v))

def mixtureDensity (weight center : I → ℝ) (v x : ℝ) : ℝ :=
  ∑ i, gaussianComponent (weight i) (center i) v x

def mixtureDerivative (weight center : I → ℝ) (v x : ℝ) : ℝ :=
  ∑ i, gaussianComponent (weight i) (center i) v x * (-(x-center i)/v)

  (∑ i, center i*gaussianComponent (weight i) (center i) v x)/mixtureDensity weight center v x

theorem mixtureDensity_pos (weight center : I → ℝ) (v x : ℝ) (hw : ∀ i, 0 < weight i) :
    0 < mixtureDensity weight center v x :=
  Finset.sum_pos (fun i _ => mul_pos (hw i) (Real.exp_pos _)) Finset.univ_nonempty

/-- Differentiate the actual Gaussian density, not an assumed score field. -/
theorem mixtureDensity_hasDeriv (weight center : I → ℝ) (v x : ℝ) (hv : 0 < v) :
    HasDerivAt (mixtureDensity weight center v) (mixtureDerivative weight center v x) x := by
  apply HasDerivAt.fun_sum
  intro i _
  change HasDerivAt (fun z => weight i*Real.exp (-((z-center i)^2)/(2*v)))
    (weight i*Real.exp (-((x-center i)^2)/(2*v)) * (-(x-center i)/v)) x
  have hd : HasDerivAt (fun z : ℝ => -((z-center i)^2)/(2*v)) (-(x-center i)/v) x := by
    convert (((hasDerivAt_id' x).sub_const (center i)).fun_pow 2).fun_neg.div_const (2*v) using 1
    field_simp [ne_of_gt hv]
    ring
  convert hd.exp.const_mul (weight i) using 1
  ring

theorem mixtureLog_hasDeriv (weight center : I → ℝ) (v x : ℝ)
    (hv : 0 < v) (hw : ∀ i, 0 < weight i) :
    HasDerivAt (fun z => Real.log (mixtureDensity weight center v z))
      (mixtureDerivative weight center v x/mixtureDensity weight center v x) x :=
  (mixtureDensity_hasDeriv weight center v x hv).log (ne_of_gt (mixtureDensity_pos weight center v x hw))

    posteriorMean weight center v x = x+v*deriv (fun z => Real.log (mixtureDensity weight center v z)) x := by
  rw [(mixtureLog_hasDeriv weight center v x hv hw).deriv]
  have hs : mixtureDerivative weight center v x =
      -(1/v)*(x*mixtureDensity weight center v x-
        ∑ i, center i*gaussianComponent (weight i) (center i) v x) := by
    unfold mixtureDerivative mixtureDensity
    rw [Finset.mul_sum, ← Finset.sum_sub_distrib, Finset.mul_sum]
    apply Finset.sum_congr rfl
    intro i _
    ring
  rw [hs]
  unfold posteriorMean
  field_simp [ne_of_gt hv, ne_of_gt (mixtureDensity_pos weight center v x hw)]
  ring

/-- A finite probability vector, permitting zero probabilities. -/
structure FiniteDistribution (α : Type*) [Fintype α] where
  mass : α → ℝ
  nonneg : ∀ i, 0 ≤ mass i
  total : ∑ i, mass i = 1

variable {α : Type*} [Fintype α]

/-- Natural-log Shannon entropy; `negMulLog 0 = 0`. -/
def shannonEntropy (p : FiniteDistribution α) : ℝ := ∑ i, Real.negMulLog (p.mass i)

lemma FiniteDistribution.mass_le_one (p : FiniteDistribution α) (i : α) : p.mass i ≤ 1 := by
  rw [← p.total]
  exact Finset.single_le_sum (fun j _ => p.nonneg j) (Finset.mem_univ i)

theorem shannonEntropy_nonneg (p : FiniteDistribution α) : 0 ≤ shannonEntropy p := by
  exact Finset.sum_nonneg fun i _ => Real.negMulLog_nonneg (p.nonneg i) (p.mass_le_one i)

/-- Extract finite masses from an actual probability measure. -/
def FiniteDistribution.ofProbabilityMeasure [MeasurableSpace α] [MeasurableSingletonClass α]
    (P : ProbabilityMeasure α) : FiniteDistribution α where
  mass i := (P.toMeasure {i}).toReal
  nonneg _ := ENNReal.toReal_nonneg
  total := by
    rw [← ENNReal.toReal_sum (fun i _ => measure_ne_top P.toMeasure {i})]
    simp

/-- Entropy upper bound: no strictly-positive-support assumption is needed. -/
theorem shannonEntropy_le_log_card [Nonempty α] (p : FiniteDistribution α) :
    shannonEntropy p ≤ Real.log (Fintype.card α) := by
  let n : ℝ := Fintype.card α
  have hn : 0 < n := by dsimp [n]; exact_mod_cast Fintype.card_pos
  have hw : ∑ _ : α, (1/n) = 1 := by simp [n, ne_of_gt hn]
  have hJ := Real.concaveOn_negMulLog.le_map_sum (t := Finset.univ)
    (w := fun _ : α => (1/n)) (p := p.mass)
    (fun _ _ => le_of_lt (one_div_pos.mpr hn)) hw (fun i _ => p.nonneg i)
  have hleft : (∑ i : α, (1/n) • Real.negMulLog (p.mass i)) = (1/n)*shannonEntropy p := by
    simp [shannonEntropy, smul_eq_mul, Finset.mul_sum]
  have hright : (∑ i : α, (1/n) • p.mass i) = 1/n := by
    simp [smul_eq_mul, ← Finset.mul_sum, p.total]
  rw [hleft, hright] at hJ
  have hlog : Real.negMulLog (1/n) = (1/n)*Real.log n := by
    simp [Real.negMulLog_def]
  rw [hlog] at hJ
  exact (mul_le_mul_iff_right₀ (one_div_pos.mpr hn)).mp hJ
end LeanML.Information

/-- Finite KL on strictly positive probability vectors; supports are explicit in theorems. -/
def finiteKL (q p : FiniteDistribution H) : ℝ := ∑ h, q.mass h*Real.log (q.mass h/p.mass h)

/-- The change-of-measure variational inequality used in PAC-Bayesian arguments.
This theorem alone is not a high-probability PAC-Bayes sample bound.

    (hq : ∀ h, 0 < q.mass h) (hp : ∀ h, 0 < p.mass h) :
    (∑ h, q.mass h*φ h) ≤ finiteKL q p+Real.log (∑ h, p.mass h*Real.exp (φ h)) := by
  have hj : ∀ h, 0 < p.mass h*Real.exp (φ h) := fun h => mul_pos (hp h) (Real.exp_pos _)
  have bound := finiteELBO_le_logEvidence (fun h => p.mass h*Real.exp (φ h)) q hj hq
  have he : finiteELBO (fun h => p.mass h*Real.exp (φ h)) q =
      (∑ h, q.mass h*φ h)-finiteKL q p := by
    unfold finiteELBO finiteKL
    rw [← Finset.sum_sub_distrib]
    apply Finset.sum_congr rfl
    intro h _
    rw [Real.log_div (ne_of_gt (hj h)) (ne_of_gt (hq h)),
      Real.log_mul (ne_of_gt (hp h)) (Real.exp_ne_zero _), Real.log_exp,
      Real.log_div (ne_of_gt (hq h)) (ne_of_gt (hp h))]
    ring
  rw [he] at bound
  linarith

/-- Finite Gibbs inequality is a corollary of the same variational principle. -/
theorem finiteKL_nonneg (q p : FiniteDistribution H)
    (hq : ∀ h, 0 < q.mass h) (hp : ∀ h, 0 < p.mass h) : 0 ≤ finiteKL q p := by
  simpa [p.total] using changeOfMeasure q p (fun _ => 0) hq hp

/-- A typed differentiable node with its actual Fréchet derivative. -/
structure DifferentiableNode (E F : Type*)
    [NormedAddCommGroup E] [NormedSpace ℝ E]
    [NormedAddCommGroup F] [NormedSpace ℝ F] where
  eval : E → F
  derivative : E → E →L[ℝ] F
  correct : ∀ x, HasFDerivAt eval (derivative x) x

/-- Composition preserves the derivative certificate. -/
def DifferentiableNode.comp (second : DifferentiableNode F G) (first : DifferentiableNode E F) :
    DifferentiableNode E G where
  eval := second.eval ∘ first.eval
  derivative x := (second.derivative (first.eval x)).comp (first.derivative x)
  correct x := (second.correct (first.eval x)).comp x (first.correct x)

/-- Shared-input branching is a typed DAG primitive. -/
def DifferentiableNode.fork (left : DifferentiableNode E F) (right : DifferentiableNode E G) :
    DifferentiableNode E (F × G) where
  eval x := (left.eval x, right.eval x)
  derivative x := (left.derivative x).prod (right.derivative x)
  correct x := (left.correct x).prodMk (right.correct x)

/-- Reverse propagation of a continuous-linear cotangent through one node. -/
def DifferentiableNode.reverse (node : DifferentiableNode E F) (x : E)
    (cotangent : F →L[ℝ] ℝ) : E →L[ℝ] ℝ := cotangent.comp (node.derivative x)

/-- The propagated cotangent is the derivative of the scalarized output. -/
theorem DifferentiableNode.reverse_correct (node : DifferentiableNode E F) (x : E)
    (cotangent : F →L[ℝ] ℝ) :
    HasFDerivAt (fun y => cotangent (node.eval y)) (node.reverse x cotangent) x :=
  cotangent.hasFDerivAt.comp x (node.correct x)

/-- A reverse pass traverses compositions in reverse evaluation order. -/
theorem DifferentiableNode.reverse_comp (second : DifferentiableNode F G)
    (first : DifferentiableNode E F) (x : E) (cotangent : G →L[ℝ] ℝ) :
    (second.comp first).reverse x cotangent =
      first.reverse x (second.reverse (first.eval x) cotangent) := rfl
end LeanML.Optimization

theorem DifferentiableNode.reverse_fork (left : DifferentiableNode E F)

  congr 1
  simp

/-- First-order characterization of differentiable convex objectives. -/
def FirstOrderConvex (f : E → ℝ) (g : E → E) : Prop :=
  ∀ x y, f x + inner ℝ (g x) (y-x) ≤ f y

def gradientStep (g : E → E) (η : ℝ) (x : E) : E := x-η • g x

def gradientIterate (g : E → E) (η : ℝ) (x₀ : E) : ℕ → E
  | 0 => x₀
  | k+1 => gradientStep g η (gradientIterate g η x₀ k)

omit [CompleteSpace E] in
@[simp] theorem gradientIterate_zero (g : E → E) (η : ℝ) (x₀ : E) :
    gradientIterate g η x₀ 0 = x₀ := rfl

/-- Loss descent for any positive step no larger than `1/L`. -/
theorem gradientStep_descent {f : E → ℝ} {g : E → E} {L η : ℝ}
    (hf : IsLSmooth f g L) (hη : 0 < η) (hηL : η*L ≤ 1) (x : E) :
    f (gradientStep g η x) ≤ f x-η/2*‖g x‖^2 := by
  have h := descent_lemma hf x (gradientStep g η x)
  simp only [gradientStep, sub_sub_cancel_left, inner_neg_right, inner_smul_right,
    real_inner_self_eq_norm_sq, norm_neg, norm_smul, Real.norm_eq_abs,
    abs_of_pos hη, mul_pow] at h
  have hprod := mul_nonneg (sub_nonneg.mpr hηL) (mul_nonneg (le_of_lt hη) (sq_nonneg ‖g x‖))
  dsimp only [gradientStep]
  nlinarith

/-- The distance potential controls the next objective gap. -/
theorem gradientStep_gap {f : E → ℝ} {g : E → E} {L η : ℝ}
    (hf : IsLSmooth f g L) (hc : FirstOrderConvex f g)
    (hη : 0 < η) (hηL : η*L ≤ 1) (x z : E) :
    2*η*(f (gradientStep g η x)-f z) ≤ ‖x-z‖^2-‖gradientStep g η x-z‖^2 := by
  have hd := gradientStep_descent hf hη hηL x
  have hcx := hc x z
  have hn : ‖gradientStep g η x-z‖^2 =
      ‖x-z‖^2-2*η*inner ℝ (g x) (x-z)+η^2*‖g x‖^2 := by
    have heq : gradientStep g η x-z = (x-z)-η • g x := by
      dsimp [gradientStep]; abel
    rw [heq, norm_sub_sq_real, inner_smul_right, real_inner_comm (x-z) (g x),
      norm_smul, Real.norm_eq_abs, abs_of_pos hη, mul_pow]
    ring
  have hi : inner ℝ (g x) (z-x) = -inner ℝ (g x) (x-z) := by
    rw [← inner_neg_right, neg_sub]
  rw [hi] at hcx
  have hm := mul_le_mul_of_nonneg_left (show f (gradientStep g η x)-f z ≤
      inner ℝ (g x) (x-z)-η/2*‖g x‖^2 by linarith) (show 0 ≤ 2*η by positivity)
  nlinarith

/-- Convex smooth GD has the explicit `O(1/k)` last-iterate objective bound.
No minimizer-existence assumption is hidden: `z` is any comparison point. -/
theorem gradientDescent_rate {f : E → ℝ} {g : E → E} {L η : ℝ}
    (hf : IsLSmooth f g L) (hc : FirstOrderConvex f g)
    (hη : 0 < η) (hηL : η*L ≤ 1) (x₀ z : E) (k : ℕ) (hk : 0 < k) :
    f (gradientIterate g η x₀ k)-f z ≤ ‖x₀-z‖^2/(2*η*k) := by
  let x := gradientIterate g η x₀
  have hmono : Antitone (fun i => f (x i)) := by
    apply antitone_nat_of_succ_le
    intro i
    have hd := gradientStep_descent hf hη hηL (x i)
    change f (gradientStep g η (x i)) ≤ f (x i)
    nlinarith [sq_nonneg ‖g (x i)‖]
  have hpotential : ∀ n : ℕ,
      2*η*(n:ℝ)*(f (x n)-f z) ≤ ‖x₀-z‖^2-‖x n-z‖^2 := by
    intro n
    induction n with
    | zero => simp [x]
    | succ n ih =>
      have hstep := gradientStep_gap hf hc hη hηL (x n) z
      have horder := hmono (Nat.le_succ n)
      have hweight : 0 ≤ 2*η*(n:ℝ) := by positivity
      have hm := mul_le_mul_of_nonneg_left (sub_le_sub_right horder (f z)) hweight
      change 2*η*(f (x (n+1))-f z) ≤ ‖x n-z‖^2-‖x (n+1)-z‖^2 at hstep
      push_cast
      nlinarith
  apply (le_div_iff₀ (show 0 < 2*η*(k:ℝ) by positivity)).mpr
  nlinarith [hpotential k, sq_nonneg ‖x k-z‖]
end LeanML.Optimization

/-- Strong convexity gives the gradient-dominance estimate used in linear convergence. -/
theorem stronglyConvex_gap_bound {f : E → ℝ} {g : E → E} {μ : ℝ}
    (hf : IsStronglyConvex f g μ) (x z : E) :
    2*μ*(f x-f z) ≤ ‖g x‖^2 := by
  have hm := hf.lower_model x z
  have hi := neg_le_of_abs_le (abs_real_inner_le_norm (g x) (z-x))
  have hs := sq_nonneg (μ*‖z-x‖-‖g x‖)
  have hmul := mul_le_mul_of_nonneg_left hm (show 0 ≤ 2*μ by linarith [hf.constant_pos])
  have hip := mul_le_mul_of_nonneg_left hi (show 0 ≤ 2*μ by linarith [hf.constant_pos])
  nlinarith

/-- Linear convergence in objective value for strongly convex smooth objectives.
This states a loss bound, separately from the stronger squared-distance bound.
-/
theorem gradientDescent_linear_rate {f : E → ℝ} {g : E → E} {L μ : ℝ}
    (hf : IsLSmooth f g L) (hstrong : IsStronglyConvex f g μ) (hμL : μ ≤ L)
    (x₀ z : E) (k : ℕ) :
    f (gradientIterate g (1/L) x₀ k)-f z ≤ (1-μ/L)^k*(f x₀-f z) := by
  have hL : 0 < L := hf.constant_pos
  have hfactor : 0 ≤ 1-μ/L := by
    apply sub_nonneg.mpr
    exact (div_le_one hL).mpr hμL
  have hstep : ∀ x, f (gradientStep g (1/L) x)-f z ≤ (1-μ/L)*(f x-f z) := by
    intro x
    have hd := gradient_descent_step_decreases hf x
    have hg := stronglyConvex_gap_bound hstrong x z
    have hscaled := mul_le_mul_of_nonneg_left hg (show 0 ≤ 1/(2*L) by positivity)
    have halg : (1/(2*L))*(2*μ*(f x-f z)) = (μ/L)*(f x-f z) := by ring
    rw [halg] at hscaled
    dsimp [gradientStep]
    nlinarith
  induction k with
  | zero => simp [gradientIterate]
  | succ k ih =>
    calc
      f (gradientIterate g (1/L) x₀ (k+1))-f z ≤
          (1-μ/L)*(f (gradientIterate g (1/L) x₀ k)-f z) := hstep _
      _ ≤ (1-μ/L)*((1-μ/L)^k*(f x₀-f z)) := mul_le_mul_of_nonneg_left ih hfactor
      _ = (1-μ/L)^(k+1)*(f x₀-f z) := by ring
end LeanML.Optimization

/-- Connect Mathlib's convexity definition to the first-order gradient characterization. -/
theorem firstOrderConvex_of_convexOn {f : E → ℝ} {g : E → E}
    (hc : ConvexOn ℝ Set.univ f) (hg : ∀ x, HasGradientAt f (g x) x) :
    FirstOrderConvex f g := by
  intro x y
  let φ : ℝ → ℝ := fun t => f (x+t • (y-x))
  have hcφ : ConvexOn ℝ Set.univ φ := by
    simpa [φ, Function.comp_def, AffineMap.lineMap_apply_module', add_comm] using
      hc.comp_affineMap (AffineMap.lineMap x y)
  have hline : HasDerivAt (fun t : ℝ => x+t • (y-x)) (y-x) 0 := by
    simpa only [one_smul] using ((hasDerivAt_id' (0:ℝ)).smul_const (y-x)).const_add x
  have hφ : HasDerivAt φ (inner ℝ (g x) (y-x)) 0 := by
    have hh := (hg x).hasFDerivAt
    have hh' : HasFDerivAt f (InnerProductSpace.toDual ℝ E (g x)) (x+(0:ℝ) • (y-x)) := by

      hh'.comp_hasDerivAt 0 hline
  have hs := hcφ.le_slope_of_hasDerivAt (Set.mem_univ (0:ℝ))
    (Set.mem_univ (1:ℝ)) zero_lt_one hφ
  simp [slope_def_field, φ] at hs
  linarith

/-- A feasible point in the adjoint range is the minimum-norm interpolant. -/
theorem adjoint_interpolant_pythagoras (A : E →L[ℝ] F) (u y : F)
    (hu : A (A.adjoint u) = y) (v : E) (hv : A v = y) :
    ‖v‖^2 = ‖A.adjoint u‖^2+‖v-A.adjoint u‖^2 := by
  have hzero : A (v-A.adjoint u) = 0 := by rw [map_sub, hv, hu, sub_self]

    rw [A.adjoint_inner_left, hzero, inner_zero_right]
  have hn := norm_add_sq_real (A.adjoint u) (v-A.adjoint u)
  have he : A.adjoint u+(v-A.adjoint u) = v := by abel
  rw [he, hi] at hn
  nlinarith

    (hu : A (A.adjoint u) = y) (v : E) (hv : A v = y) : ‖A.adjoint u‖ ≤ ‖v‖ := by
  have h := adjoint_interpolant_pythagoras A u y hu v hv
  exact (sq_le_sq₀ (norm_nonneg _) (norm_nonneg _)).mp (by nlinarith [sq_nonneg ‖v-A.adjoint u‖])

      A.adjoint (u-η • (A (A.adjoint u)-y)) := by

/-- PL concerns a genuine gradient supplied by `IsLSmooth`; convexity is not assumed. -/
structure PolyakLojasiewicz (f : E → ℝ) (g : E → E) (μ optimum : ℝ) : Prop where
  constant_pos : 0 < μ
  lower_bound : ∀ x, optimum ≤ f x
  dominance : ∀ x, 2*μ*(f x-optimum) ≤ ‖g x‖^2

omit [CompleteSpace E] [InnerProductSpace ℝ E] in
/-- Under PL, stationary points attain the stated global optimum. -/
theorem PolyakLojasiewicz.stationary_global {f : E → ℝ} {g : E → E} {μ optimum : ℝ}
    (h : PolyakLojasiewicz f g μ optimum) (x : E) (hx : g x = 0) : f x = optimum := by
  have hd := h.dominance x
  rw [hx, norm_zero, zero_pow (by decide : 2 ≠ 0)] at hd
  nlinarith [h.lower_bound x, h.constant_pos]

/-- Linear convergence of loss values under PL, with no convexity hypothesis. -/
theorem pl_gradientDescent_rate {f : E → ℝ} {g : E → E} {L μ optimum : ℝ}
    (hf : IsLSmooth f g L) (hpl : PolyakLojasiewicz f g μ optimum) (hμL : μ ≤ L)
    (x₀ : E) (k : ℕ) :
    f (gradientIterate g (1/L) x₀ k)-optimum ≤ (1-μ/L)^k*(f x₀-optimum) := by
  have hL := hf.constant_pos
  have hfactor : 0 ≤ 1-μ/L := sub_nonneg.mpr ((div_le_one hf.constant_pos).mpr hμL)
  have hstep : ∀ x, f (gradientStep g (1/L) x)-optimum ≤ (1-μ/L)*(f x-optimum) := by
    intro x
    have hd := gradient_descent_step_decreases hf x
    have hp := mul_le_mul_of_nonneg_left (hpl.dominance x)
      (show 0 ≤ 1/(2*L) by positivity)
    have he : (1/(2*L))*(2*μ*(f x-optimum)) = (μ/L)*(f x-optimum) := by ring
    rw [he] at hp
    dsimp [gradientStep]
    nlinarith
  induction k with
  | zero => simp [gradientIterate]
  | succ k ih =>
    calc
      f (gradientIterate g (1/L) x₀ (k+1))-optimum ≤
          (1-μ/L)*(f (gradientIterate g (1/L) x₀ k)-optimum) := hstep _
      _ ≤ (1-μ/L)*((1-μ/L)^k*(f x₀-optimum)) := mul_le_mul_of_nonneg_left ih hfactor
      _ = (1-μ/L)^(k+1)*(f x₀-optimum) := by ring
end LeanML.Optimization

/-! # Smooth objectives
The gradient condition is essential: Lipschitzness of an unrelated vector field
cannot imply a bound on the objective. All derivatives below are genuine Fréchet derivatives.
-/
noncomputable section
open Set InnerProductSpace

variable {E : Type*} [NormedAddCommGroup E] [InnerProductSpace ℝ E] [CompleteSpace E]

/-- A differentiable objective with a positive global Lipschitz bound on its gradient. -/
structure IsLSmooth (f : E → ℝ) (g : E → E) (L : ℝ) : Prop where
  constant_pos : 0 < L
  hasGradient : ∀ x, HasGradientAt f (g x) x
  lipschitz : ∀ x y, ‖g x - g y‖ ≤ L * ‖x - y‖

/-- First-order strong convexity, with a specified genuine gradient. -/
structure IsStronglyConvex (f : E → ℝ) (g : E → E) (μ : ℝ) : Prop where
  constant_pos : 0 < μ
  hasGradient : ∀ x, HasGradientAt f (g x) x
  lower_model : ∀ x y, f x + inner ℝ (g x) (y-x) + μ/2 * ‖y-x‖^2 ≤ f y

/-- Descent lemma, proved by monotonicity of the quadratic-corrected line restriction. -/
theorem descent_lemma {f : E → ℝ} {g : E → E} {L : ℝ}
    (hf : IsLSmooth f g L) (x y : E) :
    f y ≤ f x + inner ℝ (g x) (y-x) + L/2 * ‖y-x‖^2 := by
  let d := y-x
  let φ := fun t : ℝ => f (x+t • d) - f x - t * inner ℝ (g x) d - L/2*t^2*‖d‖^2
  let φ' := fun t : ℝ => inner ℝ (g (x+t • d)) d - inner ℝ (g x) d - L*t*‖d‖^2
  have hφ : ∀ t, HasDerivAt φ (φ' t) t := by
    intro t
    have hline : HasDerivAt (fun s : ℝ => x+s • d) d t := by
      simpa only [one_smul] using ((hasDerivAt_id' t).smul_const d).const_add x
    have hcomp : HasDerivAt (fun s : ℝ => f (x+s • d))
        (inner ℝ (g (x+t • d)) d) t := by
      simpa [Function.comp_def, InnerProductSpace.toDual_apply_apply] using
        HasFDerivAt.comp_hasDerivAt t (hf.hasGradient (x+t • d)).hasFDerivAt hline
    have hquad : HasDerivAt (fun s : ℝ => L/2*s^2*‖d‖^2) (L*t*‖d‖^2) t := by
      convert (((hasDerivAt_id' t).fun_pow 2).const_mul (L/2)).mul_const (‖d‖^2) using 1; ring
    simpa only [φ, φ', one_mul] using
      ((hcomp.sub_const (f x)).fun_sub ((hasDerivAt_id' t).mul_const (inner ℝ (g x) d))).fun_sub hquad
  have hanti : AntitoneOn φ (Icc (0:ℝ) 1) := by
    apply antitoneOn_of_hasDerivWithinAt_nonpos (convex_Icc _ _)
      (fun t _ => (hφ t).continuousAt.continuousWithinAt)
      (fun t _ => (hφ t).hasDerivWithinAt)
    intro t ht
    have ht0 : 0 ≤ t := (interior_subset ht).1
    have hnorm : ‖g (x+t • d)-g x‖ ≤ L*t*‖d‖ := by
      simpa [norm_smul, Real.norm_eq_abs, abs_of_nonneg ht0, mul_assoc] using
        hf.lipschitz (x+t • d) x
    have hinner := (real_inner_le_norm (g (x+t • d)-g x) d).trans
      (mul_le_mul_of_nonneg_right hnorm (norm_nonneg d))
    rw [inner_sub_left] at hinner
    dsimp [φ']
    nlinarith
  have h := hanti (show (0:ℝ) ∈ Icc (0:ℝ) 1 by simp)
    (show (1:ℝ) ∈ Icc (0:ℝ) 1 by simp) zero_le_one
  have hxy : x+d = y := by dsimp [d]; abel
  dsimp [φ] at h
  simp only [zero_smul, add_zero, zero_mul, sub_self, one_smul,
    one_mul, one_pow, mul_one, hxy] at h
  dsimp [d] at h
  nlinarith

/-- One gradient step of size `1/L` decreases the loss. -/
theorem gradient_descent_step_decreases {f : E → ℝ} {g : E → E} {L : ℝ}
    (hf : IsLSmooth f g L) (x : E) :
    f (x-(1/L) • g x) ≤ f x - (1/(2*L))*‖g x‖^2 := by
  have h := descent_lemma hf x (x-(1/L) • g x)
  have hL : L ≠ 0 := ne_of_gt hf.constant_pos
  have hnonneg : 0 ≤ 1/L := le_of_lt (one_div_pos.mpr hf.constant_pos)
  simp only [sub_sub_cancel_left, inner_neg_right, inner_smul_right,
    real_inner_self_eq_norm_sq, norm_neg, norm_smul, Real.norm_eq_abs,
    abs_of_nonneg hnonneg, mul_pow] at h
  convert h using 1; field_simp; ring
end LeanML.Optimization

variable {S A : Type*} [Fintype S] [Fintype A] [Nonempty A]

/-- A finite discounted MDP with normalized transition probabilities. -/
structure FiniteMDP (S A : Type*) [Fintype S] where
  reward : S → A → ℝ
  transition : S → A → FiniteDistribution S
  discount : ℝ≥0
  discount_lt_one : discount < 1

def actionValue (M : FiniteMDP S A) (V : S → ℝ) (s : S) (a : A) : ℝ :=
  M.reward s a+(M.discount:ℝ)*(∑ t, (M.transition s a).mass t*V t)

def bellman (M : FiniteMDP S A) (V : S → ℝ) (s : S) : ℝ :=
  Finset.univ.sup' Finset.univ_nonempty (actionValue M V s)

/-- Finite maxima cannot amplify a uniform coordinatewise perturbation. -/
theorem max_abs_sub_le (f g : A → ℝ) (C : ℝ) (h : ∀ a, |f a-g a| ≤ C) :
    |Finset.univ.sup' Finset.univ_nonempty f-Finset.univ.sup' Finset.univ_nonempty g| ≤ C := by
  apply abs_le.mpr
  have hfg : Finset.univ.sup' Finset.univ_nonempty f ≤
      Finset.univ.sup' Finset.univ_nonempty g+C := by
    apply Finset.sup'_le
    intro a _
    have h₁ := (abs_le.mp (h a)).2
    have h₂ := Finset.le_sup' g (Finset.mem_univ a)
    linarith
  have hgf : Finset.univ.sup' Finset.univ_nonempty g ≤
      Finset.univ.sup' Finset.univ_nonempty f+C := by
    apply Finset.sup'_le
    intro a _
    have h₁ := (abs_le.mp (h a)).1
    have h₂ := Finset.le_sup' f (Finset.mem_univ a)
    linarith
  constructor <;> linarith

/-- Averaging by a probability vector is nonexpansive in the supremum norm. -/
theorem transition_nonexpansive (p : FiniteDistribution S) (U V : S → ℝ) :
    |(∑ t, p.mass t*U t)-(∑ t, p.mass t*V t)| ≤ dist U V := by
  rw [← Finset.sum_sub_distrib]
  have he : ∀ t, p.mass t*U t-p.mass t*V t = p.mass t*(U t-V t) := by intro t; ring
  simp_rw [he]
  calc
    |∑ t, p.mass t*(U t-V t)| ≤ ∑ t, |p.mass t*(U t-V t)| := Finset.abs_sum_le_sum_abs _ _
    _ ≤ ∑ t, p.mass t*dist U V := by
      apply Finset.sum_le_sum
      intro t _
      rw [abs_mul, abs_of_nonneg (p.nonneg t)]
      apply mul_le_mul_of_nonneg_left _ (p.nonneg t)
      simpa only [Real.dist_eq] using dist_le_pi_dist U V t
    _ = dist U V := by rw [← Finset.sum_mul, p.total, one_mul]

/-- Bellman optimality is a strict contraction for discount < 1. -/
theorem bellman_contracting (M : FiniteMDP S A) : ContractingWith M.discount (bellman M) := by
  refine ⟨M.discount_lt_one, LipschitzWith.of_dist_le_mul ?_⟩
  intro U V
  apply (dist_pi_le_iff (mul_nonneg M.discount.coe_nonneg dist_nonneg)).mpr
  intro s
  rw [Real.dist_eq]
  apply max_abs_sub_le
  intro a
  have h := mul_le_mul_of_nonneg_left (transition_nonexpansive (M.transition s a) U V)
    M.discount.coe_nonneg
  dsimp only [actionValue]
  simpa only [add_sub_add_left_eq_sub, ← mul_sub, abs_mul, abs_of_nonneg M.discount.coe_nonneg] using h

/-- The optimality operator has one and only one value-function fixed point. -/
theorem bellman_exists_unique (M : FiniteMDP S A) : ∃! V, bellman M V = V := by
  let h := bellman_contracting M
  exact ⟨ContractingWith.fixedPoint _ h, h.fixedPoint_isFixedPt, fun _ hv => h.fixedPoint_unique hv⟩

/-- Value iteration converges from every finite initial value function. -/
theorem valueIteration_converges (M : FiniteMDP S A) (V₀ : S → ℝ) :
    Filter.Tendsto (fun n => (bellman M)^[n] V₀) Filter.atTop
      (nhds (ContractingWith.fixedPoint (bellman M) (bellman_contracting M))) :=
  (bellman_contracting M).tendsto_iterate_fixedPoint V₀
end LeanML.Reinforcement

variable {G : Type*} [Fintype G]

def meanFrequency (weights : FiniteDistribution G) (frequency : G → ℝ) : ℝ :=
  ∑ g, weights.mass g*frequency g

def groupedBrier (weights : FiniteDistribution G) (forecast frequency : G → ℝ) : ℝ :=
  ∑ g, weights.mass g*(frequency g*(forecast g-1)^2+(1-frequency g)*(forecast g)^2)

def reliability (weights : FiniteDistribution G) (forecast frequency : G → ℝ) : ℝ :=
  ∑ g, weights.mass g*(forecast g-frequency g)^2

def resolution (weights : FiniteDistribution G) (frequency : G → ℝ) : ℝ :=
  ∑ g, weights.mass g*(frequency g-meanFrequency weights frequency)^2

/-- Exact finite-group calibration/resolution/uncertainty decomposition.
The identity is algebraic; interpreting frequencies as probabilities additionally
requires that they belong to [0,1]. It does not assume forecasts are calibrated.
-/
theorem brier_decomposition (weights : FiniteDistribution G) (forecast frequency : G → ℝ) :
    groupedBrier weights forecast frequency =
      meanFrequency weights frequency*(1-meanFrequency weights frequency)-
      resolution weights frequency+reliability weights forecast frequency := by
  let m := meanFrequency weights frequency
  have hp : ∀ g, weights.mass g*(frequency g*(forecast g-1)^2+(1-frequency g)*(forecast g)^2) =
      weights.mass g*(forecast g-frequency g)^2-weights.mass g*(frequency g-m)^2+
      weights.mass g*frequency g-2*m*(weights.mass g*frequency g)+m^2*weights.mass g := by
    intro g
    ring
  unfold groupedBrier
  simp_rw [hp]
  simp only [Finset.sum_add_distrib, Finset.sum_sub_distrib, ← Finset.mul_sum]
  rw [weights.total]
  change reliability weights forecast frequency-resolution weights frequency+
    m-2*m*m+m^2*1 = m*(1-m)-resolution weights frequency+reliability weights forecast frequency
  ring
end LeanML.Reliability

/-! # Unmasked self-attention
Tokens are finite indexed vectors. The permutation acts on both query and key
indices. Positional encodings and causal masks are deliberately outside this theorem.
-/
noncomputable section
open scoped BigOperators
namespace LeanML.Representation
variable {N D : Type*} [Fintype N] [Fintype D]

def dot (x y : D → ℝ) : ℝ := ∑ d, x d * y d

def softmax (s : N → ℝ) (i : N) : ℝ := Real.exp (s i) / ∑ j, Real.exp (s j)

/-- Reindexing every input reindexes softmax, including its normalizer. -/
theorem softmax_permute (s : N → ℝ) (p : Equiv.Perm N) (i : N) :
    softmax (s ∘ p) i = softmax s (p i) := by
  unfold softmax
  simp only [Function.comp_apply]
  rw [Equiv.sum_comp p (fun j => Real.exp (s j))]

theorem softmax_pos [Nonempty N] (s : N → ℝ) (i : N) : 0 < softmax s i := by
  unfold softmax
  apply div_pos (Real.exp_pos _)
  exact Finset.sum_pos (fun j _ => Real.exp_pos _) Finset.univ_nonempty

theorem softmax_sum [Nonempty N] (s : N → ℝ) : ∑ i, softmax s i = 1 := by
  have h : (∑ j, Real.exp (s j)) ≠ 0 := ne_of_gt
    (Finset.sum_pos (fun j _ => Real.exp_pos (s j)) Finset.univ_nonempty)
  simp [softmax, div_eq_mul_inv, ← Finset.sum_mul, h]

/-- Scaled dot-product attention after arbitrary shared tokenwise projections. -/
def attention (scale : ℝ) (Q K V : N → D → ℝ) (i : N) (d : D) : ℝ :=
  ∑ j, softmax (fun k => dot (Q i) (K k) / scale) j * V j d

/-- Attention commutes with any token permutation applied to queries, keys and values. -/
theorem attention_permute (scale : ℝ) (Q K V : N → D → ℝ)
    (p : Equiv.Perm N) (i : N) (d : D) :
    attention scale (Q ∘ p) (K ∘ p) (V ∘ p) i d = attention scale Q K V (p i) d := by
  unfold attention
  have h : (fun k => dot ((Q ∘ p) i) ((K ∘ p) k) / scale) =
      (fun k => dot (Q (p i)) (K k) / scale) ∘ p := rfl
  rw [h]
  simp_rw [softmax_permute]
  exact Equiv.sum_comp p (fun j => softmax (fun k => dot (Q (p i)) (K k)/scale) j * V j d)

/-- Self-attention accepts shared projections; this is the architecture-level equivariance. -/
def selfAttention (scale : ℝ) (q k v : (D → ℝ) → D → ℝ) (X : N → D → ℝ) :=
  attention scale (q ∘ X) (k ∘ X) (v ∘ X)

theorem selfAttention_permute (scale : ℝ) (q k v : (D → ℝ) → D → ℝ)
    (X : N → D → ℝ) (p : Equiv.Perm N) :
    selfAttention scale q k v (X ∘ p) = selfAttention scale q k v X ∘ p := by
  funext i d
  exact attention_permute scale (q ∘ X) (k ∘ X) (v ∘ X) p i d
end LeanML.Representation

/-- Row-vector multiplication by a shared projection matrix. -/
def linearProjection (W : D → D → ℝ) (x : D → ℝ) (j : D) : ℝ := ∑ i, x i*W i j

/-- The matrix self-attention architecture, with the usual square-root head scaling. -/
def matrixSelfAttention (WQ WK WV : D → D → ℝ) (X : N → D → ℝ) : N → D → ℝ :=
  selfAttention (Real.sqrt (Fintype.card D)) (linearProjection WQ)
    (linearProjection WK) (linearProjection WV) X

    (X : N → D → ℝ) (p : Equiv.Perm N) :
    matrixSelfAttention WQ WK WV (X ∘ p) = matrixSelfAttention WQ WK WV X ∘ p :=
  selfAttention_permute _ _ _ _ X p

/-- Uniform approximation by finite-width shallow networks on a specified compact domain. -/
def UniformApproximation (input : ℕ) (K : Set (Fin input → ℝ)) (σ : ℝ → ℝ) : Prop :=
  ∀ f : (Fin input → ℝ) → ℝ, ContinuousOn f K → ∀ ε : ℝ, 0 < ε →
    ∃ width : ℕ, ∃ net : ShallowNetwork input width,
      ∀ x ∈ K, |net.eval σ x-f x| < ε

/-- The approximation property restricts to smaller domains when the same
continuous target is available on the original domain. -/
theorem approximation_restrict {input : ℕ} {K S : Set (Fin input → ℝ)} {σ : ℝ → ℝ}
    (h : UniformApproximation input K σ) (hSK : S ⊆ K)
    (f : (Fin input → ℝ) → ℝ) (hf : ContinuousOn f K) (ε : ℝ) (hε : 0 < ε) :
    ∃ width : ℕ, ∃ net : ShallowNetwork input width, ∀ x ∈ S, |net.eval σ x-f x| < ε := by
  obtain ⟨width, net, hnet⟩ := h f hf ε hε
  exact ⟨width, net, fun x hx => hnet x (hSK hx)⟩
end LeanML.Representation

variable {G : Type*} [Group G] [Fintype G]

def leftTranslate (a : G) (f : G → ℝ) (x : G) : ℝ := f (a⁻¹*x)

/-- Finite-group convolution with the signal in the first factor. -/
def groupConvolution (f k : G → ℝ) (x : G) : ℝ := ∑ y, f y*k (y⁻¹*x)

/-- Noncommutative finite-group equivariance follows by Haar/counting reindexing. -/
theorem groupConvolution_equivariant (a : G) (f k : G → ℝ) :
    groupConvolution (leftTranslate a f) k = leftTranslate a (groupConvolution f k) := by
  funext x
  unfold groupConvolution leftTranslate
  rw [← Equiv.sum_comp (Equiv.mulLeft a) (fun y => f (a⁻¹*y)*k (y⁻¹*x))]
  simp only [Equiv.coe_mulLeft, inv_mul_cancel_left, mul_inv_rev, mul_assoc]
end LeanML.Representation

variable {G : Type*} [Group G] [Fintype G]

noncomputable def identitySpike : G → ℝ := by
  classical
  exact fun g => if g=1 then 1 else 0

/-- Finite-group classification: every linear translation-equivariant operator is a convolution. -/
theorem equivariant_linear_is_convolution (T : (G → ℝ) →ₗ[ℝ] (G → ℝ))
    (hT : ∀ a f, T (leftTranslate a f) = leftTranslate a (T f)) :
    ∃ k : G → ℝ, ∀ f, T f = groupConvolution f k := by
  classical
  refine ⟨T identitySpike, ?_⟩
  intro f
  have hf : f = ∑ g : G, f g • leftTranslate g (identitySpike : G → ℝ) := by
    funext x
    simp [leftTranslate, identitySpike, inv_mul_eq_one, eq_comm]
  funext x
  calc
    T f x = T (∑ g : G, f g • leftTranslate g (identitySpike : G → ℝ)) x := congrArg (fun h => T h x) hf
    _ = ∑ g, f g*T (leftTranslate g (identitySpike : G → ℝ)) x := by
      simp only [map_sum, map_smul, Finset.sum_apply, Pi.smul_apply, smul_eq_mul]
    _ = groupConvolution f (T identitySpike) x := by
      unfold groupConvolution
      apply Finset.sum_congr rfl
      intro g _
      rw [hT]
      rfl
end LeanML.Representation

  intro j _

/-- A dense affine layer followed by a coordinatewise activation. -/
structure DenseLayer (input output : ℕ) where
  weight : Matrix (Fin output) (Fin input) ℝ
  bias : Fin output → ℝ

def DenseLayer.eval {input output : ℕ} (layer : DenseLayer input output)
    (σ : ℝ → ℝ) (x : Fin input → ℝ) : Fin output → ℝ :=
  fun i => σ ((∑ j, layer.weight i j*x j)+layer.bias i)

/-- A scalar network with one hidden layer of finite width. -/
structure ShallowNetwork (input width : ℕ) where
  hidden : DenseLayer input width
  outputWeight : Fin width → ℝ
  outputBias : ℝ

def ShallowNetwork.eval {input width : ℕ} (net : ShallowNetwork input width)
    (σ : ℝ → ℝ) (x : Fin input → ℝ) : ℝ :=
  (∑ i, net.outputWeight i*net.hidden.eval σ x i)+net.outputBias

theorem DenseLayer.continuous_eval {input output : ℕ} (layer : DenseLayer input output)
    {σ : ℝ → ℝ} (hσ : Continuous σ) : Continuous (layer.eval σ) := by
  apply continuous_pi
  intro i
  apply hσ.comp
  apply Continuous.add _ continuous_const
  exact continuous_finsetSum _ fun j _ => continuous_const.mul (continuous_apply j)

theorem ShallowNetwork.continuous_eval {input width : ℕ} (net : ShallowNetwork input width)
    {σ : ℝ → ℝ} (hσ : Continuous σ) : Continuous (net.eval σ) := by
  apply Continuous.add _ continuous_const
  apply continuous_finsetSum
  intro i _
  exact continuous_const.mul ((continuous_apply i).comp (net.hidden.continuous_eval hσ))
end LeanML.Representation

namespace LeanML.Representation
/-- Feedforward networks with checked input/output dimensions at every composition. -/
inductive MLP : ℕ → ℕ → Type
  | layer {input output : ℕ} (dense : DenseLayer input output) : MLP input output
  | comp {input hidden output : ℕ} (first : MLP input hidden) (second : MLP hidden output) :
      MLP input output

noncomputable def MLP.eval {input output : ℕ} (σ : ℝ → ℝ) :
    MLP input output → (Fin input → ℝ) → Fin output → ℝ
  | .layer dense => dense.eval σ
  | .comp first second => second.eval σ ∘ first.eval σ

theorem MLP.continuous_eval {input output : ℕ} (net : MLP input output)
    {σ : ℝ → ℝ} (hσ : Continuous σ) : Continuous (net.eval σ) := by
  induction net with
  | layer dense => exact dense.continuous_eval hσ
  | comp first second ihfirst ihsecond => exact ihsecond.comp ihfirst
end LeanML.Representation

variable {X Y Z : Type*} [PseudoMetricSpace X] [PseudoMetricSpace Y] [PseudoMetricSpace Z]

/-- A map together with its certified global Lipschitz bound. -/
structure LipschitzLayer (X Y : Type*) [PseudoMetricSpace X] [PseudoMetricSpace Y] where
  eval : X → Y
  bound : ℝ≥0
  certified : LipschitzWith bound eval

def LipschitzLayer.comp (second : LipschitzLayer Y Z) (first : LipschitzLayer X Y) :
    LipschitzLayer X Z where
  eval := second.eval ∘ first.eval
  bound := second.bound * first.bound
  certified := second.certified.comp first.certified

def evaluateLayers : List (LipschitzLayer X X) → X → X
  | [] => id
  | layer :: rest => evaluateLayers rest ∘ layer.eval

def layerProduct : List (LipschitzLayer X X) → ℝ≥0
  | [] => 1
  | layer :: rest => layerProduct rest * layer.bound

/-- Depth multiplies the certified layer constants. -/
theorem network_lipschitz (layers : List (LipschitzLayer X X)) :
    LipschitzWith (layerProduct layers) (evaluateLayers layers) := by
  induction layers with
  | nil => exact LipschitzWith.id
  | cons layer rest ih => exact ih.comp layer.certified

/-- A nonexpansive network cannot increase metric perturbations. -/
theorem perturbation_nonexpansive (layers : List (LipschitzLayer X X))
    (h : layerProduct layers ≤ 1) (x y : X) :
    dist (evaluateLayers layers x) (evaluateLayers layers y) ≤ dist x y := by
  have hb := (network_lipschitz layers).dist_le_mul x y
  have hreal : (layerProduct layers : ℝ) ≤ 1 := by exact_mod_cast h
  exact hb.trans (by simpa using mul_le_mul_of_nonneg_right hreal dist_nonneg)
end LeanML.Robustness

/-- DEQ update; the activation's Lipschitz hypothesis is indispensable. -/
def deqUpdate (σ : E → E) (W : E →L[ℝ] E) (x : E) (h : E) : E := σ (W h+x)

omit [CompleteSpace E] in
theorem deq_contracting (σ : E → E) (W : E →L[ℝ] E) (x : E)
    (hσ : LipschitzWith 1 σ) (hW : ‖W‖₊ < 1) :
    ContractingWith ‖W‖₊ (deqUpdate σ W x) := by
  refine ⟨hW, ?_⟩
  have hw : LipschitzWith ‖W‖₊ (W : E → E) :=
    ContinuousLinearMap.lipschitzWith_of_opNorm_le (f := W) le_rfl
  apply LipschitzWith.of_dist_le_mul
  intro a b
  dsimp only [deqUpdate]
  have hs := hσ.dist_le_mul (W a+x) (W b+x)
  simp only [NNReal.coe_one, one_mul, dist_add_right] at hs
  exact hs.trans (hw.dist_le_mul a b)

/-- Banach gives a unique equilibrium on the complete hidden-state space. -/
theorem deq_exists_unique (σ : E → E) (W : E →L[ℝ] E) (x : E)
    (hσ : LipschitzWith 1 σ) (hW : ‖W‖₊ < 1) : ∃! h, deqUpdate σ W x h = h := by
  let hc := deq_contracting σ W x hσ hW
  exact ⟨ContractingWith.fixedPoint _ hc, hc.fixedPoint_isFixedPt, fun _ hz => hc.fixedPoint_unique hz⟩

/-- Picard iteration converges to that equilibrium. -/
theorem deq_iterates_converge (σ : E → E) (W : E →L[ℝ] E) (x start : E)
    (hσ : LipschitzWith 1 σ) (hW : ‖W‖₊ < 1) :
    Filter.Tendsto (fun n => (deqUpdate σ W x)^[n] start) Filter.atTop
      (nhds (ContractingWith.fixedPoint (deqUpdate σ W x) (deq_contracting σ W x hσ hW))) :=
  (deq_contracting σ W x hσ hW).tendsto_iterate_fixedPoint start
end LeanML.Robustness

variable {X C : Type*} [PseudoMetricSpace X]

/-- Strict winning scores avoid ambiguity from ties in an argmax. -/
def StrictWinner (scores : X → C → ℝ) (x : X) (winner : C) : Prop :=
  ∀ competitor, competitor ≠ winner → scores x competitor < scores x winner

/-- Pairwise decision margins survive perturbations smaller than their Lipschitz budget. -/
theorem margin_stable (scores : X → C → ℝ) (K : ℝ≥0)
    (hLip : ∀ i, LipschitzWith K (fun x => scores x i)) (x y : X) (a b : C)
    (hm : 2*(K:ℝ)*dist y x < scores x a-scores x b) : scores y b < scores y a := by
  have ha := abs_le.mp (show |scores y a-scores x a| ≤ (K:ℝ)*dist y x by
    simpa only [Real.dist_eq] using (hLip a).dist_le_mul y x)
  have hb := abs_le.mp (show |scores y b-scores x b| ≤ (K:ℝ)*dist y x by
    simpa only [Real.dist_eq] using (hLip b).dist_le_mul y x)
  linarith

/-- The usual certified radius, with the indispensable positive Lipschitz constant. -/
theorem certified_radius (scores : X → C → ℝ) (K : ℝ≥0)
    (hLip : ∀ i, LipschitzWith K (fun x => scores x i)) (x y : X) (a b : C)
    (hK : 0 < (K:ℝ)) (r : ℝ) (hr : r < (scores x a-scores x b)/(2*(K:ℝ)))
    (hy : dist y x ≤ r) : scores y b < scores y a := by
  apply margin_stable scores K hLip x y a b
  have hm := (lt_div_iff₀ (show 0 < 2*(K:ℝ) by positivity)).mp hr
  have hd := mul_le_mul_of_nonneg_left hy (show 0 ≤ 2*(K:ℝ) by positivity)
  nlinarith

/-- Multiclass robustness requires a positive margin against EVERY competing class. -/
theorem strictWinner_stable (scores : X → C → ℝ) (K : ℝ≥0)
    (hLip : ∀ i, LipschitzWith K (fun x => scores x i)) (x y : X) (winner : C)
    (margin : ℝ) (hmargin : ∀ j, j ≠ winner → margin ≤ scores x winner-scores x j)
    (hbudget : 2*(K:ℝ)*dist y x < margin) : StrictWinner scores y winner := by
  intro j hj
  exact margin_stable scores K hLip x y winner j (hbudget.trans_le (hmargin j hj))

omit [PseudoMetricSpace X] in
/-- A classifier that actually selects a maximum must return the unique strict winner. -/
theorem classifier_eq_of_strictWinner (scores : X → C → ℝ) (classify : X → C)
    (hmax : ∀ z i, scores z i ≤ scores z (classify z)) (y : X) (c : C)
    (hc : StrictWinner scores y c) : classify y = c := by
  by_contra h
  exact (not_lt_of_ge (hmax y c)) (hc (classify y) h)
end LeanML.Robustness

variable {E F : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E]

/-- Operator-norm weight error gives an input-uniform output error bound. -/
theorem quantization_error (W Q : E →L[ℝ] F) (x : E) (Δ B : ℝ)
    (hΔ : ‖W-Q‖ ≤ Δ) (hB : ‖x‖ ≤ B) : ‖W x-Q x‖ ≤ Δ*B := by
  have hnonneg : 0 ≤ Δ := (norm_nonneg (W-Q)).trans hΔ
  calc
    ‖W x-Q x‖ = ‖(W-Q) x‖ := by rw [sub_apply]
    _ ≤ ‖W-Q‖*‖x‖ := (W-Q).le_opNorm x
    _ ≤ Δ*B := mul_le_mul hΔ hB (norm_nonneg _) hnonneg

/-- An activation multiplies quantization error by its Lipschitz constant. -/
theorem activated_quantization_error (W Q : E →L[ℝ] F) (σ : F → F) (K : ℝ≥0)
    (hσ : LipschitzWith K σ) (x : E) (Δ B : ℝ)
    (hΔ : ‖W-Q‖ ≤ Δ) (hB : ‖x‖ ≤ B) :
    ‖σ (W x)-σ (Q x)‖ ≤ (K:ℝ)*(Δ*B) := by
  exact (show ‖σ (W x)-σ (Q x)‖ ≤ (K:ℝ)*‖W x-Q x‖ by
    simpa only [dist_eq_norm] using hσ.dist_le_mul (W x) (Q x)).trans
      (mul_le_mul_of_nonneg_left (quantization_error W Q x Δ B hΔ hB) K.coe_nonneg)

/-- The square-root row factor is valid for a bound on EACH output coordinate. -/
theorem euclidean_coordinate_bound {I : Type*} [Fintype I]
    (e : EuclideanSpace ℝ I) (C : ℝ) (hC : 0 ≤ C) (he : ∀ i, |e i| ≤ C) :
    ‖e‖ ≤ Real.sqrt (Fintype.card I)*C := by
  have hs : ∀ i, (e i)^2 ≤ C^2 := by
    intro i
    have h := (sq_le_sq₀ (abs_nonneg (e i)) hC).mpr (he i)
    simpa only [sq_abs] using h
  apply (sq_le_sq₀ (norm_nonneg e) (mul_nonneg (Real.sqrt_nonneg _) hC)).mp
  rw [mul_pow, Real.sq_sqrt (Nat.cast_nonneg _), EuclideanSpace.real_norm_sq_eq]
  simpa using! Finset.sum_le_sum (fun i _ => hs i)

/-- Row vectors of an entrywise matrix error in Euclidean coordinates. -/
def matrixRowError {I J : Type*} [Fintype J] (W Q : I → J → ℝ) (i : I) : EuclideanSpace ℝ J :=
  WithLp.toLp 2 (fun j => W i j-Q i j)

def matrixOutputError {I J : Type*} [Fintype I] [Fintype J]
    (W Q : I → J → ℝ) (x : EuclideanSpace ℝ J) : EuclideanSpace ℝ I :=
  WithLp.toLp 2 (fun i => inner ℝ (matrixRowError W Q i) x)

/-- Correct dimension dependence for a uniform PER-ENTRY weight error.
The factor is sqrt(rows)*sqrt(columns)*Delta*B, not merely sqrt(rows)*Delta*B.
-/
theorem entrywise_quantization_bound {I J : Type*} [Fintype I] [Fintype J]
    (W Q : I → J → ℝ) (x : EuclideanSpace ℝ J) (Δ B : ℝ)
    (hΔ : 0 ≤ Δ) (hentry : ∀ i j, |W i j-Q i j| ≤ Δ) (hB : ‖x‖ ≤ B) :
    ‖matrixOutputError W Q x‖ ≤ Real.sqrt (Fintype.card I)*(Real.sqrt (Fintype.card J)*Δ*B) := by
  have hB0 := (norm_nonneg x).trans hB
  apply euclidean_coordinate_bound _ _ (by positivity)
  intro i
  change |inner ℝ (matrixRowError W Q i) x| ≤ Real.sqrt (Fintype.card J)*Δ*B
  have hrow : ‖matrixRowError W Q i‖ ≤ Real.sqrt (Fintype.card J)*Δ := by
    apply euclidean_coordinate_bound _ Δ hΔ
    intro j
    exact hentry i j
  exact (abs_real_inner_le_norm _ _).trans
    (mul_le_mul hrow hB (norm_nonneg x) (by positivity))

/-- A 1×2 row of unit coefficient errors shows why a column factor cannot be omitted. -/
theorem entrywise_bound_needs_dimension : ¬ (2:ℝ) ≤ Real.sqrt 2 := by
  intro h
  nlinarith [Real.sq_sqrt (show (0:ℝ) ≤ 2 by norm_num), Real.sqrt_nonneg (2:ℝ)]
end LeanML.Robustness

# LeanML

Biblioteca de fundamentos matemáticos de aprendizado de máquina em **Lean 4 / Mathlib**.
A entrada pública é [`LeanML.lean`](LeanML.lean). As provas são verificadas pelo kernel;
nenhum teorema pendente é introduzido como axioma.

```sh
source scripts/activate.sh
lake exe cache get
bash scripts/check.sh
python3 scripts/export_code.py
```

Instale [elan](https://github.com/leanprover/elan) antes de executar esses comandos.
O projeto fixa Lean **v4.36.0-rc2** e o commit do Mathlib em `lakefile.lean` e
`lake-manifest.json`. O Lean 4.15 do roteiro inicial é incompatível com essa revisão.
Para compilar as dependências a partir dos fontes, use `lake --no-cache build`.
Para rematerializar dependências sem baixar automaticamente o cache, use
`MATHLIB_NO_CACHE_ON_UPDATE=1 lake update`. O cache oficial requer `cache.mathlib.org`.

| Área | Resultados e interfaces |
|---|---|
| Informação | Entropia finita, KL de medidas, processamento de dados, desigualdade de informação |
| Otimização | Suavidade com gradiente genuíno, descida, taxas convexas e PL, geometria de mínimos quadrados |
| Aprendizagem | Risco, Hoeffding, Azuma-Hoeffding, Chebyshev, VC, Sauer-Shelah, mudança de medida variacional |
| Representação | MLP, atenção equivariante, kernels finitos positivos, DeepSets invariantes, classificação de convoluções em grupos finitos |
| Robustez | Produto de constantes Lipschitz, margem adversarial estrita, equilíbrio de DEQs, erro de quantização |
| Generativos | Acoplamentos finitos, dualidade fraca de transporte, ELBO, Tweedie para misturas gaussianas escalares |
| Reforço | Bellman para MDPs finitos descontados, unicidade e convergência de iteração de valor |
| Privacidade | Preservação de privacidade diferencial por pós-processamento mensurável determinístico |
| Confiabilidade | Decomposição finita de Brier em calibração, resolução e incerteza |
| Jogos | Minimax de Sion na forma de ponto de sela, reutilizado do Mathlib |

Há **40 módulos**, além da entrada e três arquivos de testes. O
[inventário](docs/STATUS.md) identifica as provas completas, especializações e
pendências dos 58 tópicos adicionais solicitados. Cybenko, NTK no limite infinito,
PAC-Bayes de alta probabilidade e reversão de Anderson **ainda não estão provados**.
Uma especificação em `Prop` não demonstra o teorema que descreve.

Os limites de robustez preservam decisões sob hipóteses explícitas de margem e
Lipschitz. Eles não certificam a verdade semântica das respostas de um modelo.
A cota de erro de quantização por entrada inclui os fatores de dimensão necessários.

O [relatório de validação](docs/VALIDATION.md) descreve os comandos e o alcance dos testes.
As [regras de contribuição](CONTRIBUTING.md) exigem hipóteses explícitas e provas sem lacunas.

O código Python/Rust e a documentação sem relação com LeanML foram retirados da
árvore ativa. Permanecem recuperáveis no histórico Git; esta versão não depende deles.

[Baixar todo o código em TXT](exports/LeanML_codigo_completo.txt).
A exportação UTF-8 inclui fontes, testes, scripts, CI, configurações e documentação,
com índice de tamanhos e hashes SHA-256. Não inclui dependências de terceiros ou binários.

## Hipóteses que não podem ser omitidas

- A suavidade relaciona o gradiente à função por `HasGradientAt`, além da constante Lipschitz.
- GD exige passo positivo e `η L ≤ 1`; a taxa explícita usa `k > 0`.
- A taxa linear implementada é para o valor objetivo, não para a norma da distância.
- KL usa valores estendidos para representar corretamente casos singulares e infinitos.
- Risco populacional é uma integral de Bochner; sua interpretação estatística exige integrabilidade.
- O Hoeffding importado expressa independência e limitações quase certamente.
- Softmax usa somas não vazias nos resultados de normalização e positividade.
- Equivariança da atenção refere-se a atenção sem máscara, com projeções compartilhadas.
- A prova finita de ELBO requer massas variacionais e joint estritamente positivos.
- A definição de aproximação uniforme não é uma demonstração de densidade.

## Inventário adicional: convenções

**Completo no domínio indicado** significa prova verificada, com todas as hipóteses
no tipo. **Parcial** identifica lemas auxiliares ou um domínio menor: não significa
a demonstração do teorema amplo solicitado. **Pendente** significa que o resultado
não está formalizado nesta revisão. Reutilização de um teorema do Mathlib é atribuída
explicitamente. Os identificadores A01–A43 seguem o anexo; C01–C15 seguem a mensagem.

## Correções matemáticas e limites

- Para erro **por entrada** Δ em uma matriz com r linhas e c colunas e ‖x‖₂ ≤ B,
  provamos `‖(W−Q)x‖₂ ≤ √r √c Δ B`. A expressão `√r Δ B` exige controle da
  norma de cada linha, em vez do erro individual de cada entrada. Uma linha (1,1)
  e entrada (1,1) dão saída 2, enquanto o limite incorreto seria √2.
- Banach exige completude e contração do mapa inteiro; ‖W‖ < 1 sozinho não
  controla uma ativação arbitrária.
- PL não afirma que toda rede satisfaz PL. O teorema não transforma automaticamente
  pontos estacionários de perdas arbitrárias em ótimos globais.
- Um limite PAC-Bayes não é sempre não vacuoso. Ele exige probabilidades,
  quantificadores e hipóteses de amostragem que não podem ser omitidos.
- Equivariância não demonstra aproximação universal, e aproximação contínua em
  compactos não significa execução de todo algoritmo computável.
- Uma prova para SDE idealizada com score exato não certifica o treinamento ou a
  distribuição final de um modelo de difusão concreto.
- Nenhum destes resultados implica ausência de alucinações semânticas, segurança
  global de aviões/dispositivos médicos ou economia garantida de inferência.

Não há alegação de que os 58 tópicos estejam concluídos, nem de prioridade histórica.

#!/usr/bin/env python3
"""Reject proof holes and new axioms in active Lean sources, preserving nested comments."""
from pathlib import Path
import re
import sys


def code_only(source):
    output = []
    depth, string, escaped = 0, False, False
    i = 0
    while i < len(source):
        pair = source[i:i+2]
        c = source[i]
        if depth:
            if pair == '/-':
                depth += 1
                i += 2
            elif pair == '-/':
                depth -= 1
                i += 2
            else:
                output.append('\n' if c == '\n' else ' ')
                i += 1
        elif string:
            output.append('\n' if c == '\n' else ' ')
            if escaped:
                escaped = False
            elif c == '\\':
                escaped = True
            elif c == '"':
                string = False
            i += 1
        elif pair == '/-':
            depth = 1
            output.append('  ')
            i += 2
        elif pair == '--':
            end = source.find('\n', i)
            i = len(source) if end < 0 else end
        elif c == '"':
            string = True
            output.append(' ')
            i += 1
        else:
            output.append(c)
            i += 1
    if depth or string:
        raise ValueError('Unterminated Lean comment or string')
    return ''.join(output)


def main():
    root = Path(__file__).resolve().parent.parent
    files = [root/'LeanML.lean', *sorted((root/'LeanML').rglob('*.lean')),
             *sorted((root/'LeanMLTest').rglob('*.lean'))]
    violations = []
    for path in files:
        code = code_only(path.read_text())
        for match in re.finditer(r'\b(sorry|admit|axiom|native_decide)\b', code):
            line = code[:match.start()].count('\n')+1
            violations.append(f'{path.relative_to(root)}:{line}: {match.group()}')
    if violations:
        print('\n'.join(violations), file=sys.stderr)
        return 1
    print(f'Proof-source check: {len(files)} active Lean files, no holes or new axioms.')
    return 0


if __name__ == '__main__':
    raise SystemExit(main())

#!/usr/bin/env python3
"""Concatenate the active LeanML sources and project configuration into UTF-8 text."""
from pathlib import Path
import argparse
import hashlib


def source_files(root):
    files = [root/name for name in (
        '.gitignore', 'README.md', 'CONTRIBUTING.md', 'LeanML.lean',
        'lakefile.lean', 'lean-toolchain', 'lake-manifest.json',
    )]
    for directory, extensions in (
        ('LeanML', {'.lean'}), ('LeanMLTest', {'.lean'}),
        ('scripts', {'.py', '.sh'}), ('.github/workflows', {'.yml', '.yaml'}),
        ('docs', {'.md'}),
    ):
        files.extend(path for path in (root/directory).rglob('*')
                     if path.is_file() and path.suffix in extensions)
    return sorted(files, key=lambda path: path.relative_to(root).as_posix())


def render(root):
    files = source_files(root)
    output = [
        'LeanML — código-fonte e configuração concatenados (UTF-8)\n',
        'Este arquivo reúne o código textual; não é um executável nem um artefato binário.\n',
        'Inclui fontes ativos, testes, scripts, CI, configurações e documentação.\n',
        'O código histórico removido e as dependências do Mathlib não integram esta exportação.\n',
        'A evidência de compilação está em docs/VALIDATION.md e o escopo em docs/STATUS.md.\n\n',
        f'ARQUIVOS: {len(files)}\n\nÍNDICE (caminho | bytes | SHA-256)\n',
    ]
    contents = []
    for path in files:
        data = path.read_bytes()
        name = path.relative_to(root).as_posix()
        output.append(f'{name} | {len(data)} | {hashlib.sha256(data).hexdigest()}\n')
        contents.append((name, data.decode('utf-8')))
    for name, content in contents:
        output.extend((f'\n===== INÍCIO DO ARQUIVO: {name} =====\n', content))
        if not content.endswith('\n'):
            output.append('\n')
        output.append(f'===== FIM DO ARQUIVO: {name} =====\n')
    return ''.join(output)


def main():
    root = Path(__file__).resolve().parent.parent
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--output', type=Path, default=root/'exports/LeanML_codigo_completo.txt')
    args = parser.parse_args()
    result = render(root)
    args.output.parent.mkdir(parents=True, exist_ok=True)
    args.output.write_text(result, encoding='utf-8', newline='')
    print(f'{args.output}: {len(source_files(root))} files, {len(result.encode("utf-8"))} bytes')


if __name__ == '__main__':
    main()
